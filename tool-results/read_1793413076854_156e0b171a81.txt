   189→export default function OpenWhenSection() {
   190→  const [preset, setPreset] = useState<keyof typeof PRESETS>("romantis");
   191→  const [fromName, setFromName] = useState("");
   192→  const [toName, setToName] = useState("");
   193→  const [theme, setTheme] = useState("rose");
   194→  const [letters, setLetters] = useState<DraftLetter[]>(PRESETS.romantis.letters);
   195→  const [openIdx, setOpenIdx] = useState<number | null>(null);
   196→  const [loading, setLoading] = useState(false);
   197→  const [error, setError] = useState("");
   198→  const [link, setLink] = useState("");
   199→  const [statusUrl, setStatusUrl] = useState("");
   200→  const [copied, setCopied] = useState(false);
   201→  const resultRef = useRef<HTMLDivElement>(null);
   202→
   203→  const applyPreset = (key: keyof typeof PRESETS) => {
   204→    setPreset(key);
   205→    setLetters(PRESETS[key].letters);
   206→    setOpenIdx(null);
   207→  };
   208→
   209→  const editLetter = (idx: number, patch: Partial<DraftLetter>) => {
   210→    setLetters((ls) => ls.map((l, i) => (i === idx ? { ...l, ...patch } : l)));
   211→  };
   212→
   213→  const submit = async () => {
   214→    setError("");
   215→    if (toName.trim().length < 1) {
   216→      setError("Isi dulu nama dia ya 😊");
   217→      return;
   218→    }
   219→    const bad = letters.findIndex((l) => l.title.trim().length < 2 || l.body.trim().length < 5);
   220→    if (bad !== -1) {
   221→      setError(`Surat #${bad + 1} belum lengkap — cek judul & isinya ya.`);
   222→      return;
   223→    }
   224→    setLoading(true);
   225→    try {
   226→      const res = await fetch("/api/letters", {
   227→        method: "POST",
   228→        headers: { "Content-Type": "application/json" },
   229→        body: JSON.stringify({
   230→          fromName: fromName.trim(),
   231→          toName: toName.trim(),
   232→          theme,
   233→          letters: letters.map((l) => ({
   234→            title: l.title.trim(),
   235→            body: l.body.trim(),
   236→            unlockDays: l.unlockDays,
   237→          })),
   238→        }),
   239→      });
   240→      const data = await res.json();
   241→      if (!res.ok) throw new Error(data.error ?? "Gagal membuat surat");
   242→      setLink(`${window.location.origin}/?k=${data.slug}`);
   243→      setStatusUrl(`${window.location.origin}/?s=${data.senderKey}`);
   244→      // simpan ke daftar pribadi (sama dgn confess) biar bisa dipantau lagi
   245→      try {
   246→        const saved = JSON.parse(
   247→          localStorage.getItem("sayangin_confessions") ?? "[]"
   248→        ) as Array<Record<string, string>>;
   249→        saved.unshift({
   250→          slug: data.slug,
   251→          senderKey: data.senderKey,
   252→          toName: toName.trim(),
   253→          theme,
   254→          type: "letters",
   255→          createdAt: new Date().toISOString(),
   256→        });
   257→        localStorage.setItem(
   258→          "sayangin_confessions",
   259→          JSON.stringify(saved.slice(0, 20))
   260→        );
   261→      } catch {
   262→        /* localStorage mungkin diblokir */
   263→      }
   264→      setTimeout(
   265→        () => resultRef.current?.scrollIntoView({ behavior: "smooth", block: "center" }),
   266→        120
   267→      );
   268→    } catch (e) {
   269→      setError(e instanceof Error ? e.message : "Terjadi kesalahan. Coba lagi.");
   270→    } finally {
   271→      setLoading(false);
   272→    }
   273→  };
   274→
   275→  const copy = async () => {
   276→    try {
   277→      await navigator.clipboard.writeText(link);
   278→      setCopied(true);
   279→      setTimeout(() => setCopied(false), 2000);
   280→    } catch {
   281→      /* diabaikan */
   282→    }
   283→  };
   284→
   285→  const shareWA = () => {
   286→    const text = encodeURIComponent(
   287→      `Aku bikinin ${letters.length} surat kecil buat kamu 💌 Satu boleh dibuka tiap minggu — pelan-pelan aja bacanya ya: ${link}`
   288→    );
   289→    window.open(`https://wa.me/?text=${text}`, "_blank");
   290→  };
   291→
   292→  return (
   293→    <section id="openwhen" className="relative overflow-hidden py-20 sm:py-28">
   294→      <div className="absolute left-1/2 top-1/3 h-72 w-[720px] -translate-x-1/2 rounded-full bg-rose-200/40 blur-[110px]" />
   295→      <div className="relative mx-auto max-w-6xl px-4 sm:px-6">
   296→        <div className="grid items-center gap-14 lg:grid-cols-2">
   297→          {/* kiri: copy + form */}
   298→          <div>
   299→            <motion.div
   300→              initial={{ opacity: 0, y: 24 }}
   301→              whileInView={{ opacity: 1, y: 0 }}
   302→              viewport={{ once: true, margin: "-80px" }}
   303→              transition={{ duration: 0.6 }}
   304→            >
   305→              <span className="inline-flex items-center gap-2 rounded-full bg-rose-100 px-4 py-1.5 text-xs font-bold tracking-wide text-rose-700">
   306→                <Sparkles className="h-3.5 w-3.5" /> FITUR FAVORIT COUPLE LDR — GRATIS
   307→              </span>
   308→              <h2 className="mt-4 text-3xl font-extrabold leading-tight tracking-tight text-rose-950 sm:text-4xl">
   309→                5 surat kecil,{" "}
   310→                <span className="font-hand text-4xl text-rose-500">dibuka tiap minggu</span>{" "}
   311→                📬
   312→              </h2>
   313→              <p className="mt-3 max-w-md text-base leading-relaxed text-rose-950/70">
   314→                Kamu tulis sekali, dia buka pelan-pelan selama sebulan. Surat

   320→            <motion.div

   324→              transition={{ duration: 0.6, delay: 0.1 }}

   326→            >
