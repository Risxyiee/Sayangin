    50→/* PRNG deterministik berbasis integer (identik di Node & browser — hindari hydration mismatch) */
    51→function rnd(i: number, salt: number): number {
    52→  let h = (i * 374761393 + salt * 668265263) >>> 0;
    53→  h = (h ^ (h >>> 13)) >>> 0;
    54→  h = Math.imul(h, 1274126177) >>> 0;
    55→  h = (h ^ (h >>> 16)) >>> 0;
    56→  return h / 4294967296;
    57→}

    59→/* ---------- falling pieces (partikel CSS — hati & titik, tanpa emoji) ---------- */
    60→export function FallingPieces({ active, kind }: { active: boolean; kind: "confetti" | "hearts" }) {
    61→  const pieces = useMemo(() => {
    62→    return Array.from({ length: 30 }, (_, i) => ({
    63→      id: i,
    64→      left: rnd(i, 20) * 96,
    65→      delay: rnd(i, 21) * 1.6,
    66→      duration: 3.2 + rnd(i, 22) * 2.6,
    67→      size: 8 + rnd(i, 23) * 12,
    68→      sway: `${rnd(i, 24) * 120 - 60}px`,
    69→      isHeart: kind === "hearts" ? true : rnd(i, 26) > 0.5,
    70→      tone: kind === "confetti" ? rnd(i, 27) : rnd(i, 25),
    71→    }));
    72→  }, [kind]);
    73→
    74→  if (!active) return null;
    75→  return (
    76→    <div className="pointer-events-none fixed inset-0 z-40 overflow-hidden" aria-hidden>
    77→      {pieces.map((p) =>
    78→        p.isHeart ? (
    79→          <span
    80→            key={p.id}
    81→            className="absolute top-[-8vh]"
    82→            style={{
    83→              left: `${p.left}%`,
    84→              width: p.size,
    85→              height: p.size,
    86→              opacity: 0.5 + p.tone * 0.4,
    87→              animation: `petal-fall ${p.duration}s linear ${p.delay}s infinite`,
    88→              ["--sway" as string]: p.sway,
    89→            }}
    90→          >
    91→            <span
    92→              className="heart-shape block"
    93→              style={{
    94→                background:
    95→                  kind === "confetti"
    96→                    ? p.tone > 0.66
    97→                      ? "#e8b4bb"
    98→                      : p.tone > 0.33
    99→                        ? "#9d1c34"
   100→                        : "#fdf6f0"
   101→                    : "#c96b7d",
   102→              }}
   103→            />
   104→          </span>
   105→        ) : (
   106→          <span
   107→            key={p.id}
   108→            className="absolute top-[-8vh] rounded-full"
   109→            style={{
   110→              left: `${p.left}%`,
   111→              width: p.size * 0.4,
   112→              height: p.size * 0.4,
   113→              background: p.tone > 0.5 ? "#d9a441" : "#fdf6f0",
   114→              opacity: 0.6,
   115→              animation: `petal-fall ${p.duration}s linear ${p.delay}s infinite`,
   116→              ["--sway" as string]: p.sway,
   117→            }}
   118→          />
   119→        )
   120→      )}
   121→    </div>
   122→  );
   123→}

   193→/* ---------- fullscreen confession experience ---------- */
   194→export default function ConfessionView({
   195→  slug,
   196→  preview,
   197→  previewKey,
   198→  onClose,
   199→}: {
   200→  slug?: string;
   201→  preview?: ConfessionData;
   202→  /** senderKey dari URL (?p=) — mode pratinjau tanpa mengubah statistik */
   203→  previewKey?: string;
   204→  onClose?: () => void;
   205→}) {
   206→  const [data, setData] = useState<ConfessionData | null>(preview ?? null);
   207→  const [stage, setStage] = useState<Stage>(preview ? "closed" : "loading");
   208→  const [isPreview, setIsPreview] = useState(false);
   209→  const [reportState, setReportState] = useState<"idle" | "done">("idle");
   210→  const [typed, setTyped] = useState("");
   211→  const [answered, setAnswered] = useState<string | null>(null);
   212→  // balasan tertulis penerima (loop dua arah)
   213→  const [replyOpen, setReplyOpen] = useState(false);
   214→  const [replyText, setReplyText] = useState("");
   215→  const [replySending, setReplySending] = useState(false);
   216→  const [replyError, setReplyError] = useState("");
   217→  const [sentReply, setSentReply] = useState<string | null>(null);
   218→  const typeTimer = useRef<ReturnType<typeof setInterval> | null>(null);
   219→
   220→  useEffect(() => {
   221→    // mode pratinjau inline: data sudah diisi via useState awal
   222→    if (preview || !slug) return;
   223→    let alive = true;
   224→    fetch(`/api/confess/${slug}${previewKey ? `?p=${previewKey}` : ""}`)
   225→      .then(async (r) => {
   226→        if (!r.ok) throw new Error("not found");
   227→        const d = await r.json();
   228→        if (!alive) return;
   229→        if (d.hidden) {
   230→          setStage("hidden");
   231→          return;
   232→        }
   233→        setIsPreview(!!d.preview);
   234→        setData({
   235→          fromName: d.fromName ?? "",
   236→          toName: d.toName,
   237→          message: d.message,
   238→          theme: d.theme ?? "rose",
   239→        });
   240→        setSentReply(d.reply ?? null);
   241→        if (d.response) {
   242→          // sudah pernah jawab → langsung ke layar jawaban, jangan tanya ulang
   243→          setAnswered(d.response);
   244→          setStage(d.response === "yes" ? "yes" : "later");
   245→        } else {
   246→          setStage("closed");
   247→        }
   248→      })
   249→      .catch(() => alive && setStage("error"));
   250→    return () => {
   251→      alive = false;
   252→    };
   253→  }, [slug, preview, previewKey]);
   254→
   255→  // efek mesin ketik
   256→  useEffect(() => {
   257→    if (stage !== "typing" || !data) return;
   258→    let i = 0;
   259→    typeTimer.current = setInterval(() => {
   260→      i += 1;
   261→      setTyped(data.message.slice(0, i));
   262→      if (i >= data.message.length) {
   263→        if (typeTimer.current) clearInterval(typeTimer.current);
   264→        setTimeout(() => setStage("question"), 700);
   265→      }
   266→    }, 38);
   267→    return () => {
   268→      if (typeTimer.current) clearInterval(typeTimer.current);
   269→    };
   270→  }, [stage, data]);
   271→
   272→  const theme = CONFESS_THEMES[data?.theme ?? "rose"] ?? CONFESS_THEMES.rose;
   273→  const senderName = data?.fromName?.trim() || "Seseorang yang diam-diam perhatian";
   274→
   275→  const respond = async (answer: "yes" | "later") => {
   276→    setAnswered(answer);
   277→    if (slug && !isPreview && !preview) {
   278→      try {
   279→        await fetch(`/api/confess/${slug}/respond`, {
   280→          method: "POST",
   281→          headers: { "Content-Type": "application/json" },
   282→          body: JSON.stringify({ response: answer }),
   283→        });
   284→      } catch {
   285→        /* biarkan UI tetap jalan walau gagal simpan */
   286→      }
   287→    }
   288→    setStage(answer === "yes" ? "yes" : "later");
   289→  };
   290→
   291→  const changeAnswer = () => {
   292→    // kembali ke pilihan (mis. "later" → "yes") — surat tampil utuh
   293→    setTyped(data?.message ?? "");
   294→    setAnswered(null);
   295→    setReplyOpen(false);
   296→    setReplyError("");
   297→    setStage("question");
   298→  };
   299→
   300→  const sendReply = async () => {
   301→    if (!slug || isPreview || preview) return;
   302→    if (replyText.trim().length < 2) {
   303→      setReplyError("Tulis balasannya dulu ya (min. 2 karakter).");
   304→      return;
   305→    }
   306→    setReplySending(true);
   307→    setReplyError("");
   308→    try {
   309→      const res = await fetch(`/api/confess/${slug}/reply`, {
   310→        method: "POST",
   311→        headers: { "Content-Type": "application/json" },
   312→        body: JSON.stringify({ reply: replyText.trim() }),
   313→      });
   314→      const d = await res.json();
   315→      if (!res.ok) throw new Error(d.error ?? "Gagal mengirim balasan. Coba lagi.");
   316→      setSentReply(d.reply ?? null);
   317→      setReplyOpen(false);
   318→      setReplyText("");
   319→    } catch (e) {
   320→      setReplyError(e instanceof Error ? e.message : "Gagal mengirim balasan. Coba lagi.");
   321→    } finally {
   322→      setReplySending(false);
   323→    }
   324→  };
   325→
   326→  const sendReport = async () => {
   327→    if (!slug || reportState === "done") return;
   328→    try {
   329→      await fetch("/api/report", {
   330→        method: "POST",
   331→        headers: { "Content-Type": "application/json" },
   332→        body: JSON.stringify({ slug }),
   333→      });
   334→    } catch {
   335→      /* tetap tandai terkirim di UI — server yang hitung */
   336→    }
   337→    setReportState("done");
   338→  };
   339→
   340→  return (
   341→    <div
   342→      className={`fixed inset-0 z-[100] overflow-y-auto bg-gradient-to-b ${theme.bg}`}
   343→    >
   344→      <FallingPieces active={stage === "closed" || stage === "question"} kind="hearts" />
   345→      <FallingPieces active={stage === "yes"} kind="confetti" />
   346→
   347→      {onClose && (
   348→        <button
   349→          onClick={onClose}
   350→          aria-label="Tutup pratinjau"
   351→          className="absolute right-4 top-4 z-50 flex h-10 w-10 items-center justify-center rounded-full bg-white/10 text-white backdrop-blur transition hover:bg-white/20"
   352→        >
   353→          <X className="h-5 w-5" />
   354→        </button>
   355→      )}
   356→
   357→      {/* banner mode pratinjau (buka lewat ?p=senderKey) */}
   358→      {isPreview && !onClose && (
   359→        <div className="fixed bottom-4 left-1/2 z-50 w-[calc(100%-2rem)] max-w-sm -translate-x-1/2">
   360→          <div className="flex items-center gap-2 rounded-full border border-amber-300/40 bg-[#3a2410]/90 px-4 py-2.5 text-xs text-amber-100 shadow-xl backdrop-blur">
   361→            <span className="flex-1">
   362→              Pratinjau — hitungan &amp; status <b>tidak tercatat</b>.
   363→            </span>
   364→            <a
   365→              href={`/?s=${previewKey}`}
   366→              className="shrink-0 font-bold text-amber-300 underline underline-offset-2"
   367→            >
   368→              ke pantauan →
   369→            </a>
   370→          </div>
   371→        </div>
   372→      )}
   373→
   374→      <div className="relative z-50 mx-auto flex min-h-full w-full max-w-md flex-col items-center justify-center px-5 py-16 text-center">
   375→        {/* ------ LOADING ------ */}
   376→        {stage === "loading" && (
   377→          <div className="flex flex-col items-center gap-4 text-white">
   378→            <Loader2 className="h-8 w-8 animate-spin text-rose-300" />
   379→            <p className="text-sm text-white/70">Membuka suratnya...</p>
   380→          </div>
   381→        )}
   382→
   383→        {/* ------ ERROR ------ */}
   384→        {stage === "error" && (
   385→          <div className="rounded-lg bg-white/10 p-8 text-white backdrop-blur">
   386→            <h2 className="font-display text-xl font-bold">Suratnya tidak ketemu</h2>
   387→            <p className="mt-2 text-sm text-white/70">
   388→              Link confess-nya mungkin salah atau sudah dihapus.
   389→            </p>
   390→            <Button
   391→              className="mt-6 rounded-md bg-primary hover:bg-[#b52743]"
   392→              onClick={() => (window.location.href = "/")}
   393→            >
   394→              Ke Sayangin
   395→            </Button>
   396→          </div>
   397→        )}
   398→
   399→        {/* ------ KONTEN DILAPORKAN ------ */}
   400→        {stage === "hidden" && (
   401→          <div className="rounded-lg bg-white/10 p-8 text-white backdrop-blur">
   402→            <h2 className="font-display text-xl font-bold">Suratnya sedang ditinjau</h2>

   411→        {data && stage === "closed" && (

   413→            initial={{ opacity: 0, y: 24 }}
   414→            animate={{ opacity: 1, y: 0 }}
   415→            className="flex flex-col items-center"
   416→          >
   417→            <p className="font-hand text-3xl text-rose-200">Buat {data.toName},</p>
