    80→/* Kompres foto di browser: max 640px, JPEG q~0.72 (≤±60KB/foto) */
    81→function compressImage(file: File): Promise<string> {
    82→  return new Promise((resolve, reject) => {
    83→    const url = URL.createObjectURL(file);
    84→    const img = new Image();
    85→    img.onload = () => {
    86→      const maxW = 480;
    87→      const maxH = 640;
    88→      const scale = Math.min(maxW / img.width, maxH / img.height, 1);
    89→      const canvas = document.createElement("canvas");
    90→      canvas.width = Math.round(img.width * scale);
    91→      canvas.height = Math.round(img.height * scale);
    92→      const ctx = canvas.getContext("2d");
    93→      if (!ctx) {
    94→        URL.revokeObjectURL(url);
    95→        reject(new Error("canvas nggak tersedia"));
    96→        return;
    97→      }
    98→      ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
    99→      URL.revokeObjectURL(url);
   100→      resolve(canvas.toDataURL("image/jpeg", 0.72));
   101→    };
   102→    img.onerror = () => {
   103→      URL.revokeObjectURL(url);
   104→      reject(new Error("foto nggak bisa dibaca"));
   105→    };
   106→    img.src = url;
   107→  });
   108→}
