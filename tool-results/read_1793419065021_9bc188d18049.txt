    35→    35→function rnd(i: number, salt: number): number {
    36→    36→  let h = (i * 374761393 + salt * 668265263) >>> 0;
    37→    37→  h = (h ^ (h >>> 13)) >>> 0;
    38→    38→  h = Math.imul(h, 1274126177) >>> 0;
    39→    39→  h = (h ^ (h >>> 16)) >>> 0;
    40→    40→  return h / 4294967296;
    41→    41→}

   464→   464→      alive = false;
   465→   465→    };
   466→   466→  }, [slug, previewKey]);
   467→   467→
   468→   468→  const theme = CONFESS_THEMES[data?.theme ?? "rose"] ?? CONFESS_THEMES.rose;
   469→   469→  const senderName = data?.fromName?.trim() || "Seseorang";

   471→   471→    switch (data?.type) {
   472→   472→      case "bouquet": return "💐";

   474→   474→      case "cake": return "🎂";

   484→   484→        method: "POST",

   492→   492→    } catch {
   493→   493→      return false;
   494→   494→    }
   495→   495→  };

   497→   497→  const sendReport = async () => {
   498→   498→    if (reportState === "done") return;
   499→   499→    try {
   500→   500→      await fetch("/api/report", {
   501→   501→        method: "POST",
   502→   502→        headers: { "Content-Type": "application/json" },
   503→   503→        body: JSON.stringify({ slug }),

   519→   519→            <span aria-hidden>👀</span>

   521→   521→              Pratinjau — hitungan &amp; status <b>gak tercatat</b>.
   522→   522→            </span>

   525→   525→              className="shrink-0 font-bold text-amber-300 underline underline-offset-2"
   526→   526→            >
   527→   527→              ke pantauan →
   528→   528→            </a>

   530→   530→        </div>
