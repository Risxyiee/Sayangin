    94→function relTime(iso: string): string {
    95→  const diff = Date.now() - new Date(iso).getTime();
    96→  const m = Math.floor(diff / 60000);
    97→  if (m < 1) return "baru saja";
    98→  if (m < 60) return `${m} menit lalu`;
    99→  const h = Math.floor(m / 60);
   100→  if (h < 24) return `${h} jam lalu`;
   101→  const d = Math.floor(h / 24);
   102→  if (d < 7) return `${d} hari lalu`;
   103→  return new Date(iso).toLocaleDateString("id-ID", {
   104→    day: "numeric",
   105→    month: "short",
   106→  });
   107→}

   113→  detail,
   114→  pulse,

   121→}) {

   232→    (l) => !l.openedAt && startedMs && startedMs + l.unlockDays * DAY > now

   268→              icon={<ShieldAlert className="h-5 w-5" />}
   269→              title="Kadonya lagi ditinjau tim"
   270→              state="waiting"

   272→                <>
   273→                  Konten ini dilaporkan beberapa kali dan sementara disembunyikan
   274→                  dari penerima.{" "}
   275→                  <a
   276→                    href={`/?k=${data.slug}&p=${senderKey}`}
   277→                    className="font-semibold text-amber-300 underline decoration-amber-400/50 underline-offset-2"
   278→                  >
   279→                    lihat kadonya →
   280→                  </a>
   281→                </>
   282→              }
   283→            />

   291→            detail={
   292→              <>
   293→                Link aktif selamanya.{" "}
   294→                <a
   295→                  href={`/?k=${data.slug}&p=${senderKey}`}
   296→                  className="inline-flex items-center gap-1 font-semibold text-rose-300 underline decoration-rose-400/50 underline-offset-2 hover:text-rose-200"

   305→                  kirim ulang <Share2 className="h-3 w-3" />
   306→                </button>
   307→              </>
   308→            }
   309→          />
   310→
   311→          {/* 2. dibuka */}
   312→          {data.openedAt ? (
   313→            <Step
   314→              icon={<MailOpen className="h-5 w-5" />}
   315→              title={`Udah dibuka! (${data.views}x)`}
   316→              state="done"
   317→              detail={
   318→                data.startedAt
   319→                  ? "Dia udah mulai bacanya — surat pertama langsung kebuka."
   320→                  : "Dia udah lihat link-nya tapi belum tekan “mulai”. Sabar ya 😌"
   321→              }
   322→            />
   323→          ) : (
   324→            <Step
   325→              icon={<Mail className="h-5 w-5" />}
   326→              title="Belum dibuka"
   327→              state="waiting"
   328→              detail="Link-nya belum dibuka. Kasih tahu dia bahwa ini spesial — pelan-pelan aja. 😌"

   331→
   332→          {/* 3. progres baca */}

   337→                readCount === letters.length

   342→              pulse={unlockedCount - readCount > 0}
   343→              detail={
   344→                <>
   345→                  Perjalanan dimulai {relTime(data.startedAt!)}.
   346→                  {replyCount > 0 && (
   347→                    <> Dia sudah bales {replyCount} surat — cek di bawah! 🥹</>

   357→              icon={<Clock className="h-5 w-5" />}
   358→              title="Belum mulai baca"
   359→              state="soft"
   360→              detail="Waktu buka baru berjalan setelah dia tekan “mulai” di halamannya."
   361→            />
   362→          )}
   363→
   364→          {/* daftar per surat */}
   365→          <div className="space-y-1.5 pt-1">
   366→            {letters.map((l) => {
   367→              const unlockMs = startedMs !== null ? startedMs + l.unlockDays * DAY : null;
   368→              const isUnlocked = unlockMs !== null && unlockMs <= now;
   369→              const daysLeft = unlockMs ? Math.ceil((unlockMs - now) / 86400000) : null;

   372→                  key={l.order}

   376→                    className={`flex h-8 w-8 shrink-0 items-center justify-center rounded-full border ${

   379→                        : isUnlocked
   380→                          ? "border-rose-400/60 bg-rose-500/20 text-rose-300"
   381→                          : "border-white/15 bg-white/5 text-white/40"
   382→                    }`}
   383→                  >
   384→                    {l.openedAt ? (
   385→                      <BookOpenCheck className="h-4 w-4" />
   386→                    ) : isUnlocked ? (
   387→                      <MailOpen className="h-4 w-4" />
   388→                    ) : (
   389→                      <Lock className="h-4 w-4" />
   390→                    )}
   391→                  </span>
   392→                  <div className="min-w-0 flex-1">
   393→                    <p className="truncate text-xs font-semibold text-white/90">
   394→                      {l.order + 1}. {l.title}
   395→                    </p>
   396→                    <p className="text-[11px] text-white/45">
   397→                      {l.openedAt
   398→                        ? `dibaca ${relTime(l.openedAt)} • ${l.views}x`
   399→                        : isUnlocked
   400→                          ? "sudah bisa dibuka — belum dibaca"
   401→                          : daysLeft
   402→                            ? `terbuka dalam ${daysLeft} hari lagi`
   403→                            : "menunggu dia mulai"}
   404→                    </p>
   405→                    {l.reply && (
   406→                      <div className="mt-1.5 rounded-lg border border-emerald-400/30 bg-emerald-500/10 px-2.5 py-1.5 text-left">
   407→                        <p className="text-[9px] font-bold uppercase tracking-wide text-emerald-300">
   408→                          ✉️ dibalas {l.repliedAt ? relTime(l.repliedAt) : ""}
   409→                        </p>
   410→                        <p className="mt-0.5 whitespace-pre-wrap text-[11px] leading-4 text-emerald-100/90">
   411→                          {l.reply}
   412→                        </p>
   413→                      </div>
   414→                    )}
   415→                  </div>
   416→                </div>
   417→              );
   418→            })}
   419→          </div>
   420→        </div>
   421→      </div>
   422→
   423→      <SavedListBox saved={saved} senderKey={senderKey} listOpen={listOpen} setListOpen={setListOpen} />
   424→
   425→      <a
   426→        href="/#openwhen"
   427→        className="flex items-center justify-center gap-2 rounded-full border border-rose-400/40 bg-rose-500/10 py-3.5 text-sm font-bold text-rose-200 transition hover:bg-rose-500/20"
   428→      >
   429→        <Plus className="h-4 w-4" /> Bikin set surat lagi
   430→      </a>
   431→
   432→      <p className="pt-2 text-center text-[11px] leading-relaxed text-white/40">
   433→        Simpan link halaman ini ({typeof window !== "undefined" ? window.location.href.slice(0, 40) : ""}…)
   434→        — ini satu-satunya cara buat lihat status.
   435→      </p>
   436→    </motion.div>
   437→  );
